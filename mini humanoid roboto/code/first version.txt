#include <Servo.h>

// Create two Servo objects
Servo servo1;

// Set the pins where the servos are connected

const int servoPin5 = 9;
const int servoPin6 = 10;
int time=15;

void setup() {
  // Attach the servos to the defined pins
  servo1.attach(servoPin1);
  servo2.attach(servoPin2);
  servo3.attach(servoPin3);
  servo4.attach(servoPin4);
  foot1.attach(servoPin5);
  foot2.attach(servoPin6);

  // Initialize both servos to 90 degrees (neutral position)
  servo1.write(90);
  servo2.write(90);
  servo3.write(90);
  servo4.write(90);
  foot1.write(90);
  foot2.write(90);
  
  delay(1000); // Allow some time for servos to reach position
}
